    47→def _plan_header(plan_num, plan_title, sop_ref):
    48→    """Standard plan header: section_heading + ref_line + Purpose intro."""
    49→    story = []
    50→    story.append(section_heading(plan_num, plan_title))

    55→def _approval_block(plan_title):
    56→    """Standard approval signature block at end of each plan."""
    57→    story = []
    58→    story.append(Spacer(1, 12))
    59→    story.append(Paragraph('<b>Plan Approval</b>', s_h2))
    60→    story.append(para(
    61→        f'The undersigned approve this <b>{plan_title}</b> as a written policy of '
    62→        f'Well Spring Intervention LLC, effective upon the date of the last signature '
    63→        f'below. This plan shall be reviewed at minimum annually and updated as '
    64→        f'needed to reflect changes in CARF standards, regulatory requirements, '
    65→        f'organizational structure, or program design. The QP maintains the signed '
    66→        f'original in the facility compliance binder and provides a copy to all '
    67→        f'personnel with implementation responsibilities.'
    68→    ))
    69→    story.append(Spacer(1, 8))
    70→    sig_data = [
    71→        ['Role', 'Name (Printed)', 'Signature', 'Date'],
    72→        ['Executive Director', '', '', ''],
    73→        ['Clinical Director', '', '', ''],
    74→        ['Qualified Professional (QP)', '', '', ''],
    75→        ['Compliance Officer', '', '', ''],
    76→    ]
    77→    sig_th = ParagraphStyle('sigth', fontName=BODY_BOLD, fontSize=9.5, leading=12,
    78→                             textColor=colors.white, alignment=TA_LEFT)
    79→    sig_td = ParagraphStyle('sigtd', fontName=BODY_FONT, fontSize=9.5, leading=12,
    80→                             textColor=TEXT_PRIMARY, alignment=TA_LEFT)
    81→    sig_data_p = [[Paragraph(f'<b>{h}</b>', sig_th) for h in sig_data[0]]]
    82→    for row in sig_data[1:]:
    83→        sig_data_p.append([Paragraph(c, sig_td) for c in row])
    84→    t = Table(sig_data_p, colWidths=[0.22*AVAIL_W, 0.28*AVAIL_W, 0.32*AVAIL_W, 0.18*AVAIL_W],
    85→              hAlign='CENTER', repeatRows=1)
    86→    t.setStyle(TableStyle([
    87→        ('BACKGROUND', (0, 0), (-1, 0), HEADER_FILL),
    88→        ('GRID', (0, 0), (-1, -1), 0.4, BORDER),
    89→        ('VALIGN', (0, 0), (-1, -1), 'MIDDLE'),
    90→        ('LEFTPADDING', (0, 0), (-1, -1), 4),
    91→        ('RIGHTPADDING', (0, 0), (-1, -1), 4),
    92→        ('TOPPADDING', (0, 0), (-1, -1), 10),
    93→        ('BOTTOMPADDING', (0, 0), (-1, -1), 10),
    94→    ]))
    95→    story.append(t)
    96→    story.append(Spacer(1, 12))
    97→    return story
